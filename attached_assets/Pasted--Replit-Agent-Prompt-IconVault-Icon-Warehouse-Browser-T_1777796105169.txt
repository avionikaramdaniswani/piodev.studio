## DESIGN SYSTEM — NEO BRUTALISM

This is the most important part. Every single component MUST follow Neo Brutalism strictly.

### Color Palette
```css
--bg-primary: #FFFBF0;        /* warm off-white / cream */
--bg-secondary: #F5F0E8;      /* slightly darker cream */
--accent-yellow: #FFE034;     /* loud yellow */
--accent-pink: #FF6B9D;       /* hot pink */
--accent-blue: #4DBBFF;       /* electric blue */
--accent-green: #00E676;      /* neon green */
--accent-orange: #FF6B35;     /* vivid orange */
--text-primary: #0A0A0A;      /* near black */
--text-secondary: #3D3D3D;
--border: #0A0A0A;            /* thick black borders everywhere */
--shadow: #0A0A0A;            /* hard offset shadow, no blur */
```

### Neo Brutalism Rules (NEVER break these)
1. **Borders**: All cards, buttons, inputs → `border: 3px solid #0A0A0A`
2. **Shadows**: Hard offset shadows ONLY → `box-shadow: 4px 4px 0px #0A0A0A` (no blur radius!)
3. **Hover states**: Buttons shift on hover → `transform: translate(-2px, -2px); box-shadow: 6px 6px 0px #0A0A0A`
4. **Active/click**: Press effect → `transform: translate(2px, 2px); box-shadow: 2px 2px 0px #0A0A0A`
5. **Border radius**: Maximum 4px — mostly 0px (square corners)
6. **Typography**: Bold, chunky, oversized headings. No subtle fonts
7. **Colors**: Each section/tool gets its own loud accent color as background
8. **NO gradients, NO soft shadows, NO blur effects, NO rounded cards**

### Reusable CSS Classes to Create
```css
.nb-card { border: 3px solid #0A0A0A; box-shadow: 4px 4px 0 #0A0A0A; background: white; }
.nb-btn { border: 3px solid #0A0A0A; box-shadow: 4px 4px 0 #0A0A0A; font-weight: 800; transition: all 0.1s; cursor: pointer; }
.nb-btn:hover { transform: translate(-2px,-2px); box-shadow: 6px 6px 0 #0A0A0A; }
.nb-btn:active { transform: translate(2px,2px); box-shadow: 2px 2px 0 #0A0A0A; }
.nb-input { border: 3px solid #0A0A0A; outline: none; font-family: inherit; }
.nb-input:focus { box-shadow: 4px 4px 0 #0A0A0A; }
.nb-badge { border: 2px solid #0A0A0A; font-weight: 700; font-size: 11px; padding: 2px 8px; }
```

### Database Schema (run in Supabase SQL editor)
```sql
-- Enable UUID extension
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- Profiles table (extends auth.users)
CREATE TABLE profiles (
  id UUID REFERENCES auth.users(id) PRIMARY KEY,
  username TEXT UNIQUE NOT NULL,
  display_name TEXT,
  avatar_url TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Icons table
CREATE TABLE icons (
  id UUID DEFAULT uuid_generate_v4() PRIMARY KEY,
  name TEXT NOT NULL,
  slug TEXT UNIQUE NOT NULL,
  description TEXT,
  svg_content TEXT NOT NULL,         -- SVG stored as raw text
  category TEXT NOT NULL,
  tags TEXT[] DEFAULT '{}',
  style TEXT DEFAULT 'outline',      -- outline | filled | duotone
  author_id UUID REFERENCES profiles(id),
  downloads INTEGER DEFAULT 0,
  likes INTEGER DEFAULT 0,
  is_featured BOOLEAN DEFAULT false,
  license TEXT DEFAULT 'MIT',
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- User likes/bookmarks
CREATE TABLE icon_likes (
  user_id UUID REFERENCES profiles(id),
  icon_id UUID REFERENCES icons(id),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  PRIMARY KEY (user_id, icon_id)
);

-- Download logs
CREATE TABLE icon_downloads (
  id UUID DEFAULT uuid_generate_v4() PRIMARY KEY,
  icon_id UUID REFERENCES icons(id),
  user_id UUID REFERENCES profiles(id),
  downloaded_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- RLS Policies
ALTER TABLE profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE icons ENABLE ROW LEVEL SECURITY;
ALTER TABLE icon_likes ENABLE ROW LEVEL SECURITY;

-- Profiles: anyone can read, only owner can update
CREATE POLICY "Public profiles" ON profiles FOR SELECT USING (true);
CREATE POLICY "Update own profile" ON profiles FOR UPDATE USING (auth.uid() = id);
CREATE POLICY "Insert own profile" ON profiles FOR INSERT WITH CHECK (auth.uid() = id);

-- Icons: anyone can read
CREATE POLICY "Public icons" ON icons FOR SELECT USING (true);
CREATE POLICY "Auth users can insert" ON icons FOR INSERT WITH CHECK (auth.uid() IS NOT NULL);

-- Likes: users manage their own
CREATE POLICY "Own likes" ON icon_likes FOR ALL USING (auth.uid() = user_id);

-- Auto-create profile on signup
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO public.profiles (id, username, display_name)
  VALUES (
    NEW.id,
    SPLIT_PART(NEW.email, '@', 1),
    SPLIT_PART(NEW.email, '@', 1)
  );
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();

-- Indexes
CREATE INDEX idx_icons_category ON icons(category);
CREATE INDEX idx_icons_tags ON icons USING GIN(tags);
CREATE INDEX idx_icons_slug ON icons(slug);
```

### `downloadSVG(svgContent, filename)`
```js
export function downloadSVG(svgContent, filename) {
  const blob = new Blob([svgContent], { type: 'image/svg+xml' })
  const url = URL.createObjectURL(blob)
  const a = document.createElement('a')
  a.href = url
  a.download = `${filename}.svg`
  a.click()
  URL.revokeObjectURL(url)
}
```

### `copyToClipboard(text)`
```js
export async function copyToClipboard(text) {
  await navigator.clipboard.writeText(text)
  // trigger toast
}
```
